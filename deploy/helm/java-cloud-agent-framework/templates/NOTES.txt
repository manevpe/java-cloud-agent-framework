1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "java-cloud-agent-framework.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch its status by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "java-cloud-agent-framework.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "java-cloud-agent-framework.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "java-cloud-agent-framework.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:{{ .Values.service.targetPort }}
{{- end }}

2. Check application health:
   kubectl --namespace {{ .Release.Namespace }} exec deploy/{{ include "java-cloud-agent-framework.fullname" . }} -- \
     wget -qO- http://localhost:{{ .Values.service.targetPort }}/actuator/health

3. Secrets: this release {{ if .Values.secrets.create }}generated its own Secret from values.secrets.values (plaintext in values.yaml — fine for a first deploy, not for anything real).{{ else }}expects a pre-existing Secret named {{ include "java-cloud-agent-framework.secretName" . }} with the required keys (see values.yaml's secrets.values for the list).{{ end }}

4. Workflows: edit values.yaml's `workflows` map and run `helm upgrade` to
   add/change a workflow definition — no image rebuild required.
